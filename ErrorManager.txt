'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-04
' Parameters:    strClassName As String, strMethodName As String, lngNumber As Long, strDescription As String, varContext As ParamArray
' Returns:       Boolean
' Description:   Adds an error and optional contextual values to the internal collection.
'-------------------------------------------------------------------------------
Public Function addError(ByVal strClassName As String, _
                         ByVal strMethodName As String, _
                         ByVal lngNumber As Long, _
                         ByVal strDescription As String, _
                         ParamArray varContext() As Variant) As Boolean
    Const METHOD_NAME As String = "addError"
    Dim blnHasContext As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngContextUpper As Long
    Dim strContext As String
    Dim strEntry As String
    Dim varContextCopy As Variant

    On Error GoTo ErrorHandler

    EnsureErrors
    On Error Resume Next
    lngContextUpper = UBound(varContext)
    blnHasContext = (VBA.Err.Number = 0)
    VBA.Err.Clear
    On Error GoTo ErrorHandler
    If blnHasContext Then
        varContextCopy = varContext
        strContext = BuildContextText(varContextCopy)
    End If
    strEntry = Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbTab & _
               strClassName & vbTab & _
               strMethodName & vbTab & _
               CStr(lngNumber) & vbTab & _
               ReplaceLineBreaks(strDescription) & _
               strContext
    mErrors.Add strEntry
    addError = True

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    addError = False
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-05
' Parameters:    ---
' Returns:       ---
' Description:   Appends stored errors to a dedicated worksheet in this workbook.
'-------------------------------------------------------------------------------
Public Sub logToWorksheet()
    Const METHOD_NAME As String = "logToWorksheet"
    Dim arrEntry As Variant
    Dim arrOutput(1 To 1, 1 To 6) As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngIndex As Long
    Dim lngNextRow As Long
    Dim lngPart As Long
    Dim strContext As String
    Dim wksLog As Excel.Worksheet

    On Error GoTo ErrorHandler

    EnsureErrors
    If mErrors.Count = 0 Then GoTo ExitSub

    Set wksLog = GetOrCreateErrorLogWorksheet()
    If wksLog Is Nothing Then GoTo ExitSub

    For lngIndex = 1 To mErrors.Count
        arrEntry = Split(CStr(mErrors(lngIndex)), vbTab)
        strContext = vbNullString

        If UBound(arrEntry) >= 5 Then
            For lngPart = 5 To UBound(arrEntry)
                If Len(strContext) > 0 Then
                    strContext = strContext & " | "
                End If
                strContext = strContext & CStr(arrEntry(lngPart))
            Next lngPart
        End If

        lngNextRow = wksLog.Cells(wksLog.Rows.Count, 1).End(xlUp).Row + 1
        If lngNextRow < 2 Then lngNextRow = 2

        arrOutput(1, 1) = GetArrayValue(arrEntry, 0)
        arrOutput(1, 2) = GetArrayValue(arrEntry, 1)
        arrOutput(1, 3) = GetArrayValue(arrEntry, 2)
        arrOutput(1, 4) = GetArrayValue(arrEntry, 3)
        arrOutput(1, 5) = GetArrayValue(arrEntry, 4)
        arrOutput(1, 6) = strContext

        wksLog.Cells(lngNextRow, 1).Resize(1, 6).Value = arrOutput
    Next lngIndex

ExitSub:
    Set wksLog = Nothing
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-05
' Parameters:    ---
' Returns:       Excel.Worksheet
' Description:   Returns the error-log worksheet and creates it when it does not exist.
'-------------------------------------------------------------------------------
Private Function GetOrCreateErrorLogWorksheet() As Excel.Worksheet
    Const METHOD_NAME As String = "GetOrCreateErrorLogWorksheet"
    Dim errDescription As String
    Dim errNumber As Long
    Dim wksLog As Excel.Worksheet

    On Error GoTo ErrorHandler

    On Error Resume Next
    Set wksLog = ThisWorkbook.Worksheets(ERROR_LOG_SHEET_NAME)
    VBA.Err.Clear
    On Error GoTo ErrorHandler

    If wksLog Is Nothing Then
        Set wksLog = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        wksLog.Name = ERROR_LOG_SHEET_NAME
        wksLog.Cells(1, 1).Value = "Timestamp"
        wksLog.Cells(1, 2).Value = "Module/Class"
        wksLog.Cells(1, 3).Value = "Procedure"
        wksLog.Cells(1, 4).Value = "Error Number"
        wksLog.Cells(1, 5).Value = "Description"
        wksLog.Cells(1, 6).Value = "Context"
        wksLog.Range("A1:F1").Font.Bold = True
        wksLog.Range("A1:F1").AutoFilter
        wksLog.Columns("A").ColumnWidth = 20
        wksLog.Columns("B:C").ColumnWidth = 28
        wksLog.Columns("D").ColumnWidth = 14
        wksLog.Columns("E:F").ColumnWidth = 50
        wksLog.Columns("E:F").WrapText = True
    End If

    Set GetOrCreateErrorLogWorksheet = wksLog

ExitFunction:
    Set wksLog = Nothing
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Set GetOrCreateErrorLogWorksheet = Nothing
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-04
' Parameters:    ---
' Returns:       ---
' Description:   Appends stored errors to the configured text log folder.
'-------------------------------------------------------------------------------
Public Sub logToTextFile()
    Const METHOD_NAME As String = "logToTextFile"
    Dim errDescription As String
    Dim errNumber As Long
    Dim blnWriteHeader As Boolean
    Dim fileNumber As Long
    Dim fso As Object
    Dim lngIndex As Long
    Dim strFilePath As String
    Dim strFolderPath As String

    On Error GoTo ErrorHandler

    EnsureErrors
    If mErrors.Count = 0 Then GoTo ExitSub
    If Len(mLogFolderPath) = 0 Then GoTo ExitSub

    strFolderPath = mLogFolderPath
    If Right$(strFolderPath, 1) <> "\" Then
        strFolderPath = strFolderPath & "\"
    End If
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(strFolderPath) Then
        Err.Raise 1020, METHOD_NAME, "Configured error log folder does not exist."
    End If

    strFilePath = strFolderPath & "VBA_Error_Log_" & Format$(Date, "yyyymmdd") & ".log"
    blnWriteHeader = Not fso.FileExists(strFilePath)
    If Not blnWriteHeader Then
        blnWriteHeader = (fso.GetFile(strFilePath).Size = 0)
    End If
    fileNumber = FreeFile
    Open strFilePath For Append As #fileNumber
    If blnWriteHeader Then
        Print #fileNumber, "Timestamp" & vbTab & "Class" & vbTab & "Method" & vbTab & "ErrorNumber" & vbTab & "Description" & vbTab & "Context"
    End If
    For lngIndex = 1 To mErrors.Count
        Print #fileNumber, CStr(mErrors(lngIndex))
    Next lngIndex

ExitSub:
    On Error Resume Next
    If fileNumber > 0 Then Close #fileNumber
    On Error GoTo 0
    Set fso = Nothing
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-04
' Parameters:    arrContext As Variant
' Returns:       String
' Description:   Formats context names and values supplied through ParamArray.
'-------------------------------------------------------------------------------
Private Function BuildContextText(ByVal arrContext As Variant) As String
    Const METHOD_NAME As String = "BuildContextText"
    Dim arrNames As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngIndex As Long
    Dim strContext As String
    Dim strName As String
    Dim varItem As Variant

    On Error GoTo ErrorHandler

    lngIndex = 0
    For Each varItem In arrContext
        If lngIndex = 0 Then
            arrNames = Split(CStr(varItem), ";")
        Else
            strName = "ContextValue" & CStr(lngIndex)
            If IsArray(arrNames) Then
                If lngIndex - 1 <= UBound(arrNames) Then
                    strName = Trim$(CStr(arrNames(lngIndex - 1)))
                End If
            End If
            strContext = strContext & vbTab & strName & "=" & SafeValueToString(varItem)
        End If
        lngIndex = lngIndex + 1
    Next varItem
    BuildContextText = strContext

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    BuildContextText = vbNullString
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-04
' Parameters:    varValue As Variant
' Returns:       String
' Description:   Converts a context value to a safe single-line representation.
'-------------------------------------------------------------------------------
Private Function SafeValueToString(ByVal varValue As Variant) As String
    Const METHOD_NAME As String = "SafeValueToString"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strValue As String

    On Error GoTo ErrorHandler

    If IsError(varValue) Then
        strValue = "#ERROR"
    ElseIf IsObject(varValue) Then
        strValue = "[Object:" & TypeName(varValue) & "]"
    ElseIf IsArray(varValue) Then
        strValue = "[Array]"
    ElseIf IsEmpty(varValue) Then
        strValue = vbNullString
    Else
        strValue = CStr(varValue)
    End If
    SafeValueToString = ReplaceLineBreaks(strValue)

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    SafeValueToString = "[Unavailable]"
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-04
' Parameters:    strValue As String
' Returns:       String
' Description:   Replaces line breaks and tabs with spaces.
'-------------------------------------------------------------------------------
Private Function ReplaceLineBreaks(ByVal strValue As String) As String
    Const METHOD_NAME As String = "ReplaceLineBreaks"
    Dim errDescription As String
    Dim errNumber As Long

    On Error GoTo ErrorHandler

    strValue = Replace$(strValue, vbCrLf, " ")
    strValue = Replace$(strValue, vbCr, " ")
    strValue = Replace$(strValue, vbLf, " ")
    strValue = Replace$(strValue, vbTab, " ")
    ReplaceLineBreaks = strValue

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    ReplaceLineBreaks = strValue
    HandleInternalFailure METHOD_NAME, errNumber, errDescription
    Resume ExitFunction
End Function
